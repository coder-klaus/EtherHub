## 浅拷贝

实现方式:

1. `展开运算符`
2. `Object.assign`

```js
const user1 = {
  name: 'Klaus',
  friend: {
    name: 'Alex'
  }
}

const user2 = { ...user1 }

user1.name = 'Steven'
console.log(user2.name) // => Klaus

user1.friend.name = 'Jhon'
console.log(user2.friend.name) // => Jhon
```

```js
const user1 = {};
const user2 = {
  name: 'klaus',
  friends: ['Alex']
};

// Object.assign(target, ...sources)
// 作用：将一个或多个源对象（sources）的属性复制到目标对象（target）中。
// 参数：
//   - target: 目标对象，接收源对象的属性。
//   - sources: 一个或多个源对象，其属性将被复制到目标对象中。
// 返回值：返回修改后的目标对象（target）。
// 特性：
//   1. 该方法是纯函数，不会修改目标对象。
//   2. 对于复制的属性，有则覆盖，无则添加。
//   3. 引用类型值是浅拷贝。
//   4. 多个源对象会从左到右依次复制属性，后面的源对象会覆盖前面的同名属性。
const user3 = Object.assign({}, user2);

console.log(user1); // 输出：{}，user1 是空对象，没有被修改。
console.log(user3); // 输出：{ name: 'klaus', friends: ['Alex'] }，user3 是 user2 的浅拷贝。
```



## 深拷贝

1. `JSON.parse + JSON.stringify`
2. 手动或借助第三方库「 如`loadsh` 或 `underscore`等 」实现



### parse + stringify

```js
const user1 = {
  name: 'klaus',
  friends: ['Alex']
}

const user2 = JSON.parse(JSON.stringify(user1));

user2.name = 'klaus2';
user2.friends[0] = 'Alex2';

console.log(user1); // { name: 'klaus', friends: [ 'Alex' ] }
console.log(user2); // { name: 'klaus2', friends: [ 'Alex2' ] }
```



**局限性**

1. 借助JSON的功能，但JSON不支持`symbol`，`undefined`和`方法`  => 会被自动过滤

   ```js
   const user1 = {
     name: 'Klaus',
     friend: {
       name: 'Alex'
     },
     running() {
       console.log('running')
     },
     [Symbol()]: 'symbol',
     address: undefined
   }
   
   const user2 = JSON.parse(JSON.stringify(user1))
   
   // JSON不支持方法，undefined 和 symbol类型值
   console.log(user2) // => { name: 'Klaus', friend: { name: 'Alex' } }
   ```

2. 无法处理`循环引用` => `死循环`

   ```js
   const user1 = {
     name: 'Klaus',
     friend: {
       name: 'Alex'
     }
   }
   
   user1.user = user1
   
   const user2 = JSON.stringify(user1) // error
   ```



### 手动实现

```js
function isObject(obj) {
  return typeof obj === 'object' && obj !== null;
}

export function deepClone(obj, weakMap = new WeakMap()) {
  // 拷贝symbol类型值
  if (typeof obj === 'symbol') {
    // Symbol(undefined) 等价于 Symbol() 等价于 Symbol('')
    return Symbol(obj.description);
  }

  // 基本类型值「 除symbol类型外 」 和 函数直接返回
  if (!isObject(obj) || obj === null) {
    return obj;
  }

  // 解决循环引用
  if (weakMap.has(obj)) {
    return weakMap.get(obj);
  }

  // 特殊对象处理：通过构造函数动态创建实例 => 将基于传入的实例创建一个完全相同的实例对象
  const constructor = obj.constructor;
  if (/^(Date|RegExp)$/i.test(constructor.name) || obj instanceof Error) {
    return new constructor(obj);
  }

  // 创建克隆对象并保留原型链
  const clone = Array.isArray(obj)
    ? []
    : obj instanceof Set
    ? new Set()
    : obj instanceof Map
    ? new Map()
    : Object.create(Object.getPrototypeOf(obj));

  // 将克隆对象存入 WeakMap
  weakMap.set(obj, clone);

  // 处理 Set
  if (obj instanceof Set) {
    obj.forEach((value) => {
      clone.add(deepClone(value, weakMap));
    });
    return clone;
  }

  // 处理 Map
  if (obj instanceof Map) {
    obj.forEach((value, key) => {
      clone.set(deepClone(key, weakMap), deepClone(value, weakMap));
    });
    return clone;
  }

  // 普通对象和数组处理
  return Reflect.ownKeys(obj).reduce((acc, key) => {
    acc[typeof key === 'symbol' ? Symbol(key.description) : key] = deepClone(obj[key], weakMap);
    return acc;
  }, clone);
}

```
