React 的通信机制强调单向数据流，属性的传递方向是单向的，生命周期函数的触发也是单向的



## 父子通信

```jsx
import { memo, useState } from 'react'

const Double = memo(({x = 0}) => {
  return (
    <div>{ x * 2 }</div>
  )
})

const Footer = memo(({add}) => {
  return (
    <button onClick={add}>
      click me
    </button>
  )
})

// 父子通信 通过props，无论是父传子，还是子传父
// 当然父子通信也可以通过ref，但这并不被推荐
const App = memo(() => {
  const [x, setX] = useState(0)

  return (
    <div>
      <Double x={x} />
      <Footer add={() => setX(x + 1)} />
    </div>
  )
})

export default App
```



## 祖孙通信

新建一个上下文对象

+ 祖先组件将需要传递的属性和方法放入上下文
+ 后代组件从上下文中取出对应属性和方法并进行使用



### 类组件

```jsx
import { PureComponent, createContext } from 'react'

// 创建上下文对象
//  + 为了便于区分推荐命名格式为xxxContext
//  + 存放目录一般为 src/context
//  + 可以直接解构Provider/Customer进行使用
const CounterContext = createContext()

// 子组件
class Counter extends PureComponent {
  render() {
    return (
      // 通过消费者消费数据
      <CounterContext.Consumer>
        {
          // 通过函数方式slot来消费数据 => 返回值是需要渲染的JSX
          value => {
            return (
              <div>
                <h2>当前计数: {value.counter}</h2>
                <button onClick={value.add}>+1</button>
              </div>
            )
          }
        }
      </CounterContext.Consumer>
    )
  }
}

// 父组件
export class App extends PureComponent {
  state = {
    counter: 0
  }

  add = () => {
    this.setState({
      counter: this.state.counter + 1
    })
  }

  render() {
    return (
      // 通过生产者提供数据
      <CounterContext.Provider value={{ counter: this.state.counter, add: this.add }}>
        <Counter />
      </CounterContext.Provider>
    )
  }
}

export default App
```

### 函数组件

函数组件的消费只能通过`Provider`和`Consumer`。 并不存在静态属性`contextType`

```jsx
import { memo, createContext } from 'react'

// 无论是类组件还是函数组件都可以通过参数设置默认值
const CounterContext = createContext({ counter: 100 })

const Counter = memo(() => {
  return (
    <CounterContext.Consumer>
      {
        ctx => {
          return (
            <div>
              <div>{ JSON.stringify(ctx) }</div>
              <div>{ String(ctx === undefined) }</div>
            </div>
          )
        }
      }
    </CounterContext.Consumer>
  )
})

const App = memo(() => {
  return (
    <>
      {/* 上下文对象的默认值是设置给没有生产者的消费者的  */}
      <Counter />

      {/* 普通使用 */}
      <CounterContext.Provider value={{ counter: 190 }}>
        <Counter />
      </CounterContext.Provider>

      {/* 对于存在生产者的消费者，如果生产者没有提供value，则value的默认值为undefined */}
      {/* 此时控制台会出现红色警告 */}
      <CounterContext.Provider>
        <Counter />
      </CounterContext.Provider>
    </>
  )
})

export default App
```

```jsx
import { memo, createContext, useContext } from 'react'

const CounterContext = createContext()

const Counter = memo(() => {
  // 通过hook获取上下文对象，并解构后使用
  const { counter, add } = useContext(CounterContext)

  return (
    <div>
      <h2>当前计数: {counter}</h2>
      <button onClick={add}>+1</button>
    </div>
  )
})

const App = memo(() => {
  return (
    <>
      <CounterContext.Provider value={{ counter: 190 }}>
        <Counter />
      </CounterContext.Provider>
    </>
  )
})

export default App
```
