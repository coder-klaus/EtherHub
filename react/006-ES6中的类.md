当我们用 `new` 创建一个类的实例时，实际上会执行类中的构造函数 `constructor`。

而 `new` 的作用就是在创建实例的同时调用构造函数来初始化实例。

```js
class Person {
  // new 构造函数 => 自动执行构造函数
  // 构造函数中 this => 组件实例
  constructor(x) {
    // 通过构造函数挂载实例属性, 值可以运行时传入
    this.x = x; // 设置 x
  }

  // 类表达式挂载属性，值是固定写死的
  y = 20; // 类表达式编译时会移到构造函数中，所以也可以加分号
}

let per = new Person(10);
console.log(per.x)
console.log(per.y)
```

```jsx
class Person {
  constructor(x) {
    this.x = x;
  }

  // 往实例上挂方法 编译时会移动到构造器中
  // 所以上层作用域其实是构造器，因此函数内部this指向实例
  getX = () => this.x
}

let per = new Person(10);
console.log(per.getX())
```

```jsx
class Person {
  constructor(x) {
    // 这里是函数语句，所以可以加分号
    this.x = x;
  }

  // 这里是类语法，不要加分号 「 否则会报错 」
  y = 20

  // 往实例原型上挂方法 => 默认是不可枚举的
  getX() {
    return this.x
  }
}

// 原型上挂属性只能在类的外面挂载
Person.prototype.y = 200;

let per = new Person(10);
console.log(per.getX())
```

```jsx
class Person {
  // 静态属性
  static name = 'Person'

  // 静态方法 => 默认是不可枚举的
  static getStaticName() {
    return this.name
  }
}

console.log(Person.name) // Person
console.log(Person.getStaticName()) // Person
```



```jsx
class Person {
  constructor(x) {
    this.x = x
  }

  static getName() {
    console.log('Person')
  }

  y = 20

  getY() {
    console.log(this.y)
  }
}

// 子类通过继承机制可以访问父类的所有非私有属性和方法，包括静态方法、实例属性和原型方法。
class Student extends Person {
  // 以下是默认的构造函数
  // constructor(...args) {
  //   super(...args);
  //   // 在子类的构造函数中，必须先调用 super 方法，才能初始化 this，
  //   // 因为 this 的初始化依赖于父类的构造函数。
  //   // 调用 super 后，才可以在构造函数中或其余方法中访问 this「重要！！！」
  //   // 在调用 super 时，本质上是执行父类的构造函数，并将子类的 this 绑定到父类的构造函数上。
  //   // 构造器第一行必须调用 super，确保了可以正常初始化父类继承过来的实例属性。
  // }
}

Student.getName()
// stu的原型链 ==> stu实例对象 => Student.prototype => Person.prototype => Object.prototype
const stu = new Student()
stu.getY()
```

```jsx
class Person {
  // 私有属性必须通过类表达式声明才可以使用
  // 不能直接在构造函数中使用
  #total = 0

  constructor(x, y) {
    this.#total = x + y;
  }

  getTotal() {
    return this.#total;
  }
}

let per = new Person(10, 20);
console.log(pre.getTotal())
```
